诚实边界：两套软著的实际实现为轻量 JSON 文件持久化（Node 内置 fs 落盘，无编译依赖、开箱即用）。本页给出的是规模化后的关系型设计——用于展示数据库设计能力，支撑未来多用户/多设备场景。 Honest boundary: the shipped copyrights use lightweight JSON-file persistence (Node's built-in fs, zero dependencies). This page shows the relational design for scale — to demonstrate database skills and support future multi-user/multi-device scenarios.

关键 DDL 片段Key DDL excerpt

CREATE TABLE metrics (
  metric_key VARCHAR(32)   NOT NULL COMMENT '指标键：chlorine/hardness/ph/turbidity/temp/heavyMetal',
  name       VARCHAR(16)   NOT NULL COMMENT '中文名',
  unit       VARCHAR(16)   NULL     COMMENT '单位：mg/L/NTU/℃/ppb',
  safe_min   DECIMAL(8,2)  NULL     COMMENT '正常（绿）下界',
  safe_max   DECIMAL(8,2)  NULL     COMMENT '正常（绿）上界',
  warn_min   DECIMAL(8,2)  NULL     COMMENT '预警（黄）下界',
  warn_max   DECIMAL(8,2)  NULL     COMMENT '预警（黄）上界',
  PRIMARY KEY (metric_key)
) ENGINE=InnoDB COMMENT='水质指标字典（三档阈值）';

CREATE TABLE readings (
  id          BIGINT       NOT NULL AUTO_INCREMENT,
  device_id   VARCHAR(36)  NOT NULL,
  metric_key  VARCHAR(32)  NOT NULL,
  value       DECIMAL(10,3) NOT NULL,
  recorded_at DATETIME     NOT NULL,
  PRIMARY KEY (id),
  KEY idx_readings_lookup (device_id, metric_key, recorded_at),
  CONSTRAINT fk_reading_device FOREIGN KEY (device_id)  REFERENCES devices (id),
  CONSTRAINT fk_reading_metric FOREIGN KEY (metric_key) REFERENCES metrics (metric_key)
) ENGINE=InnoDB COMMENT='水质时序读数';
